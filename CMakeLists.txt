cmake_minimum_required(VERSION 3.16)
project(deform3d VERSION 0.13.0 LANGUAGES CXX)

add_library(deform3d INTERFACE)
add_library(deform3d::deform3d ALIAS deform3d)
target_include_directories(deform3d INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_compile_features(deform3d INTERFACE cxx_std_17)

option(DEFORM3D_BUILD_TESTS "Build standalone tests" ON)
option(DEFORM3D_BUILD_BENCH "Build CPU benchmark" ON)

install(FILES LICENSE DESTINATION share/licenses/deform3d)

include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/deform3dConfig.cmake.in
  ${CMAKE_CURRENT_BINARY_DIR}/deform3dConfig.cmake
  INSTALL_DESTINATION lib/cmake/deform3d)

  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/deform3dConfig.cmake
              ${CMAKE_CURRENT_BINARY_DIR}/deform3dConfigVersion.cmake
  DESTINATION lib/cmake/deform3d)
